        function updateDBStatus() {
            const db = JSON.parse(localStorage.getItem(DB_KEY) || '[]');
            dbCount.textContent = db.length;

            if (db.length === 0) {
                dbStatus.classList.add('empty');
                dbMessage.textContent = 'Click "Run Simulation" to start saving data';
            } else {
                dbStatus.classList.remove('empty');
                const lastRecord = db[db.length - 1];
                dbMessage.textContent = `Last saved: ${lastRecord.date} ${lastRecord.time}`;
            }
        }

        // ==========================================
        // 1. DIGITAL TWIN

        class DigitalTwin {
            constructor() {
                this.energy_usage_kwh = 450.0;
                this.water_usage_liters = 1200.0;
                this.waste_bin_level_percent = 10.0;
                this.date = 'N/A';
                this.time_str = 'N/A';
            }

            updateSensors() {
                this.energy_usage_kwh += (Math.random() * 50) - 20;
                this.water_usage_liters += (Math.random() * 130) - 50;
                this.waste_bin_level_percent += (Math.random() * 2.5) + 0.5;

                if (this.waste_bin_level_percent > 100) this.waste_bin_level_percent = 100;

                const now = new Date();
                this.date = now.toISOString().split('T')[0];
                this.time_str = now.toTimeString().split(' ')[0];
            }
        }

        // ==========================================

        const GRID_SIZE = 5;

        function generateCameraFeed() {
            const grid = [];
            for (let i = 0; i < GRID_SIZE; i++) {
                grid[i] = [];
                for (let j = 0; j < GRID_SIZE; j++) {
                    grid[i][j] = Math.random() < 0.3 ? 1 : 0;
                }
            }
            return grid;
        }

        function detectWaste(grid) {
            let wasteCount = 0;
            let output = "\n[CV] COMPUTER VISION FEED:\n   ";
            for (let i = 0; i < GRID_SIZE; i++) output += "---";
            output += "\n";

            for (let i = 0; i < GRID_SIZE; i++) {
                output += "   |";
                for (let j = 0; j < GRID_SIZE; j++) {
                    if (grid[i][j] === 1) {
                        output += " X ";
                        wasteCount++;
                    } else {
                        output += " . ";
                    }
                }
                output += "|\n";
            }
            output += "   ";
            for (let i = 0; i < GRID_SIZE; i++) output += "---";
            output += "\n";

            log(output, 'info');
            return wasteCount;
        }

        // ==========================================
        // 3. GENERATIVE AI
        // ==========================================
        function generateAIReport(twin, wasteDetected) {
            log("\n==================================================", 'info');
            log(`[AI] SUSTAINABILITY REPORT`, 'info');
            log(`Date: ${twin.date} | Time: ${twin.time_str}`, 'info');
            log("==================================================", 'info');

            if (twin.energy_usage_kwh > 480.0) {
                log(`[!] ENERGY: High usage (${twin.energy_usage_kwh.toFixed(1)} kWh)! Recommendation: Turn off unused lights in Block A.`, 'warn');
            } else {
                log(`[OK] ENERGY: Normal usage (${twin.energy_usage_kwh.toFixed(1)} kWh). Keep it up!`);
            }

            if (twin.water_usage_liters > 1300.0) {
                log(`[!] WATER: Possible leakage (${twin.water_usage_liters.toFixed(1)} L)! Recommendation: Inspect restrooms on Floor 2.`, 'error');
            } else {
                log(`[OK] WATER: Normal usage (${twin.water_usage_liters.toFixed(1)} L).`);
            }

            if (twin.waste_bin_level_percent > 80.0 || wasteDetected > 2) {
                log(`[!] WASTE: Critical (Bin: ${twin.waste_bin_level_percent.toFixed(1)}%, CV Detections: ${wasteDetected})! Recommendation: Dispatch cleaning crew immediately.`, 'error');
            } else {
                log(`[OK] WASTE: Under control (Bin: ${twin.waste_bin_level_percent.toFixed(1)}%, CV Detections: ${wasteDetected}).`);
            }

            log("==================================================\n", 'info');
        }

        // ==========================================
        // 4. WEBSITE DATABASE
        // ==========================================
        function saveToWebsiteDB(twin, wasteCount) {
            const db = JSON.parse(localStorage.getItem(DB_KEY) || '[]');

            const status = (twin.energy_usage_kwh > 480 ||
                            twin.water_usage_liters > 1300 ||
                            twin.waste_bin_level_percent > 80 ||
                            wasteCount > 2) ? 'CRITICAL' : 'OK';

            db.push({
                date: twin.date,
                time: twin.time_str,
                energy: parseFloat(twin.energy_usage_kwh.toFixed(1)),
                water: parseFloat(twin.water_usage_liters.toFixed(1)),
                waste: parseFloat(twin.waste_bin_level_percent.toFixed(1)),
                cv: wasteCount,
                status: status
            });

            localStorage.setItem(DB_KEY, JSON.stringify(db));

            // INSTANTLY update the status bar
            updateDBStatus();
        }

        function viewWebsiteDB() {
            const db = JSON.parse(localStorage.getItem(DB_KEY) || '[]');

            if (db.length === 0) {
                dbView.innerHTML = '<p style="color:var(--red); margin-top:20px;">No records in website database yet. Run the simulation first.</p>';
                return;
            }

            let html = `<h3 style="color:var(--purple); margin-top:30px;">💾 Website Database (localStorage)</h3>`;
            html += `<p>Total records: <strong>${db.length}</strong></p>`;
            html += '<table class="db-table"><thead><tr>';
            html += '<th>#</th><th>Date</th><th>Time</th><th>Energy (kWh)</th><th>Water (L)</th><th>Waste (%)</th><th>CV</th><th>Status</th>';
            html += '</tr></thead><tbody>';

            let criticalCount = 0, totalEnergy = 0, totalWater = 0;

            db.forEach((row, i) => {
                const cls = row.status === 'CRITICAL' ? 'status-critical' : 'status-ok';
                html += `<tr>
                    <td>${i + 1}</td>
                    <td>${row.date}</td>
                    <td>${row.time}</td>
                    <td>${row.energy}</td>
                    <td>${row.water}</td>
                    <td>${row.waste}</td>
                    <td>${row.cv}</td>
                    <td class="${cls}">${row.status}</td>
                </tr>`;
                if (row.status === 'CRITICAL') criticalCount++;
                totalEnergy += row.energy;
                totalWater += row.water;
            });

            html += '</tbody></table>';

            html += `
                <div class="stats-grid">
                    <div class="stat-card"><h4>Total Records</h4><p>${db.length}</p></div>
                    <div class="stat-card"><h4 style="color:var(--red);">Critical Alerts</h4><p style="color:var(--red);">${criticalCount}</p></div>
                    <div class="stat-card"><h4>Avg Energy (kWh)</h4><p>${(totalEnergy / db.length).toFixed(1)}</p></div>
                    <div class="stat-card"><h4>Avg Water (L)</h4><p>${(totalWater / db.length).toFixed(1)}</p></div>
                </div>
            `;

            dbView.innerHTML = html;
            dbView.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }

        function clearWebsiteDB() {
            if (confirm('Delete all records from the website database?')) {
                localStorage.removeItem(DB_KEY);
                dbView.innerHTML = '<p style="color:var(--green); margin-top:20px;">✅ Website database cleared.</p>';
                updateDBStatus();
                log('> Website database cleared.', 'success');
            }
        }

        function exportWebsiteDB() {
            const db = JSON.parse(localStorage.getItem(DB_KEY) || '[]');
            if (db.length === 0) {
                alert('No records to export. Run the simulation first.');
                return;
            }

            let csv = 'Date,Time,Energy_kWh,Water_L,Waste_Percent,CV_Detections,Status\n';
            db.forEach(r => {
                csv += `${r.date},${r.time},${r.energy},${r.water},${r.waste},${r.cv},${r.status}\n`;
            });

            const blob = new Blob([csv], { type: 'text/csv' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'website_campus_log.csv';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        }

        async function runSimulation() {
            if (isRunning) return;
            isRunning = true;
            runBtn.disabled = true;
            terminal.innerHTML = "> Starting AI-Powered Sustainable Campus Digital Twin...\n> (Web Edition)\n";

            const twin = new DigitalTwin();

            for (let cycle = 1; cycle <= 5; cycle++) {
                log(`\n--- Cycle ${cycle} ---`, 'info');

                twin.updateSensors();
                log(`[Sensors] Energy=${twin.energy_usage_kwh.toFixed(1)} kWh | Water=${twin.water_usage_liters.toFixed(1)} L | Waste=${twin.waste_bin_level_percent.toFixed(1)}%`);

                const grid = generateCameraFeed();
                const wasteCount = detectWaste(grid);
                log(`   -> Waste objects detected: ${wasteCount}`);

                generateAIReport(twin, wasteCount);

                saveToWebsiteDB(twin, wasteCount);
                log(`   [DB] Record saved to website database. (Total: ${JSON.parse(localStorage.getItem(DB_KEY)).length})`, 'success');

                await new Promise(r => setTimeout(r, 1500));
            }

            log("\nDemo Complete. Click '💾 View Website DB' to see all records.", 'success');
            runBtn.disabled = false;
            isRunning = false;
        }

        function downloadCFile() {
            const cCode = `#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#ifdef _WIN32
    #include <windows.h>
#else
    #include <unistd.h>
#endif

#define CSV_FILE "campus_log.csv"
#define GRID_SIZE 5

typedef struct {
    float energy_usage_kwh;
    float water_usage_liters;
    float waste_bin_level_percent;
    char date[20];
    char time_str[20];
} DigitalTwin;

void init_twin(DigitalTwin *twin) {
    twin->energy_usage_kwh = 450.0f;
    twin->water_usage_liters = 1200.0f;
    twin->waste_bin_level_percent = 10.0f;
    strcpy(twin->date, "N/A");
    strcpy(twin->time_str, "N/A");
}

void update_sensors(DigitalTwin *twin) {
    twin->energy_usage_kwh += ((float)(rand() % 50) - 20.0f);
    twin->water_usage_liters += ((float)(rand() % 130) - 50.0f);
    twin->waste_bin_level_percent += ((float)(rand() % 25) / 10.0f + 0.5f);

    if (twin->waste_bin_level_percent > 100.0f)
        twin->waste_bin_level_percent = 100.0f;

    time_t rawtime;
    struct tm *timeinfo;
    time(&rawtime);
    timeinfo = localtime(&rawtime);

    strftime(twin->date, sizeof(twin->date), "%Y-%m-%d", timeinfo);
    strftime(twin->time_str, sizeof(twin->time_str), "%H:%M:%S", timeinfo);
}

int detect_waste(int grid[GRID_SIZE][GRID_SIZE]) {
    int waste_count = 0;
    printf("\\n[CV] COMPUTER VISION FEED:\\n   ");
    for (int i = 0; i < GRID_SIZE; i++) printf("---");
    printf("\\n");

    for (int i = 0; i < GRID_SIZE; i++) {
        printf("   |");
        for (int j = 0; j < GRID_SIZE; j++) {
            if (grid[i][j] == 1) {
                printf(" X ");
                waste_count++;
            } else {
                printf(" . ");
            }
        }
        printf("|\\n");
    }
    printf("   ");
    for (int i = 0; i < GRID_SIZE; i++) printf("---");
    printf("\\n");
    return waste_count;
}

void generate_camera_feed(int grid[GRID_SIZE][GRID_SIZE]) {
    for (int i = 0; i < GRID_SIZE; i++)
        for (int j = 0; j < GRID_SIZE; j++)
            grid[i][j] = (rand() % 100 < 30) ? 1 : 0;
}

int csv_exists() {
    FILE *fp = fopen(CSV_FILE, "r");
    if (fp == NULL) return 0;
    fclose(fp);
    return 1;
}

void init_csv() {
    if (!csv_exists()) {
        FILE *fp = fopen(CSV_FILE, "w");
        if (fp == NULL) return;
        fprintf(fp, "Date,Time,Energy_kWh,Water_L,Waste_Percent,CV_Detections,Status\\n");
        fclose(fp);
        printf("[DB] Database initialized: %s\\n", CSV_FILE);
    } else {
        printf("[DB] Existing database found: %s\\n", CSV_FILE);
    }
}

void save_to_csv(DigitalTwin *twin, int waste_detected) {
    FILE *fp = fopen(CSV_FILE, "a");
    if (fp == NULL) return;

    const char *status = "OK";
    if (twin->energy_usage_kwh > 480.0f ||
        twin->water_usage_liters > 1300.0f ||
        twin->waste_bin_level_percent > 80.0f ||
        waste_detected > 2) {
        status = "CRITICAL";
    }

    fprintf(fp, "%s,%s,%.1f,%.1f,%.1f,%d,%s\\n",
            twin->date, twin->time_str,
            twin->energy_usage_kwh,
            twin->water_usage_liters,
            twin->waste_bin_level_percent,
            waste_detected,
            status);

    fclose(fp);
}

            int cv;
            char status[20];
            sscanf(line, "%*[^,],%*[^,],%f,%f,%f,%d,%19s", &e, &w, &ws, &cv, status);
            total_energy += e;
            total_water += w;
            if (strcmp(status, "CRITICAL") == 0) critical++;
        }
        count++;
    }
    printf("=======================================\\n");
    printf("[DB] Total records: %d\\n", count - 1);
    if (count > 1) {
        printf("[DB] Critical alerts: %d\\n", critical);
        printf("[DB] Average Energy: %.1f kWh\\n", total_energy / (count - 1));
        printf("[DB] Average Water: %.1f L\\n", total_water / (count - 1));
    }

    fclose(fp);
}
